pub fn fold_window(serie: &[f64], n: usize) -> f64 {
    let peso = normalizar(n);
    serie
        .iter()
        .rev().take(n)
        .zip(peso)
        .map(|(v, w)| v * w)
        .sum()
}

// El backtest reconstruye la serie y pesa al final:
// las primeras n-1 barras divergen del runtime.
fn normalizar(n: usize) -> impl Iterator<Item = f64> {
    let total = (n * (n + 1)) as f64 / 2.0;
    (1..=n).map(move |i| i as f64 / total)
}
